# 人岗匹配算法设计

## 模块概述

人岗匹配模块实现可解释的智能推荐，解决传统算法的"黑盒"问题。

### 向量相似度计算

```go
// 余弦相似度
func CosineSimilarity(a, b []float64) float64 {
    if len(a) != len(b) {
        return 0
    }
    
    var dotProduct, normA, normB float64
    for i := range a {
        dotProduct += a[i] * b[i]
        normA += a[i] * a[i]
        normB += b[i] * b[i]
    }
    
    if normA == 0 || normB == 0 {
        return 0
    }
    
    return dotProduct / (math.Sqrt(normA) * math.Sqrt(normB))
}
```

### 技能匹配算法

```go
type SkillMatcher struct {
    embeddingClient *EmbeddingClient
    skillWeights    map[string]float64
}

func (m *SkillMatcher) CalculateSkillMatch(
    talentSkills []string, 
    jobSkills []string,
) (float64, []MatchDetail) {
    var details []MatchDetail
    
    // 1. 精确匹配
    exactMatches := m.exactMatch(talentSkills, jobSkills)
    
    // 2. 语义匹配 (使用Embedding)
    semanticMatches := m.semanticMatch(talentSkills, jobSkills)
    
    // 3. 计算加权得分
    totalWeight := 0.0
    matchedWeight := 0.0
    
    for _, skill := range jobSkills {
        weight := m.getSkillWeight(skill)
        totalWeight += weight
        
        if m.isMatched(skill, exactMatches, semanticMatches) {
            matchedWeight += weight
            details = append(details, MatchDetail{
                Dimension: "skill",
                Item:      skill,
                Matched:   true,
            })
        }
    }
    
    score := matchedWeight / totalWeight
    return score, details
}
```

### 经验匹配算法

```go
func CalculateExperienceMatch(
    talentExp int, 
    jobLevel string,
) (float64, string) {
    requirements := map[string]struct{ min, ideal, max int }{
        "junior":     {0, 1, 2},
        "mid":        {2, 4, 6},
        "senior":     {5, 7, 10},
        "expert":     {8, 10, 15},
        "management": {5, 8, 15},
    }
    
    req := requirements[jobLevel]
    
    if talentExp >= req.ideal {
        return 1.0, "经验完全匹配"
    } else if talentExp >= req.min {
        return 0.8, "经验基本匹配"
    } else {
        return float64(talentExp) / float64(req.min) * 0.6, "经验略显不足"
    }
}
```

type AttributionGenerator struct {
    cozeClient      *coze.Client
    embeddingClient *EmbeddingClient
}

type AttributionReport struct {
    Summary       string            `json:"summary"`
    MatchScore    float64           `json:"match_score"`
    Dimensions    []DimensionScore  `json:"dimensions"`
    Strengths     []string          `json:"strengths"`
    Gaps          []string          `json:"gaps"`
    Recommendation string           `json:"recommendation"`
}

func (g *AttributionGenerator) Generate(
    talent TalentProfile,
    job JobProfile,
    matchResult MatchResult,
) (*AttributionReport, error) {
    // 1. 构建上下文
    context := g.buildContext(talent, job, matchResult)
    
    // 2. 调用Coze大模型生成归因报告
    prompt := g.buildPrompt(context)
    response, err := g.cozeClient.Chat(prompt)
    if err != nil {
        return nil, err
    }
    
    // 3. 解析并返回报告
    return g.parseReport(response)
}

## 性能优化

### 向量缓存

```go
type EmbeddingCache struct {
    cache *redis.Client
    ttl   time.Duration
}

func (c *EmbeddingCache) GetOrCompute(
    text string,
    compute func() ([]float64, error),
) ([]float64, error) {
    key := "embedding:" + md5Hash(text)
    
    // 尝试从缓存获取
    if cached, err := c.cache.Get(ctx, key).Bytes(); err == nil {
        var embedding []float64
        json.Unmarshal(cached, &embedding)
        return embedding, nil
    }
    
    // 计算并缓存
    embedding, err := compute()
    if err != nil {
        return nil, err
    }
    
    data, _ := json.Marshal(embedding)
    c.cache.Set(ctx, key, data, c.ttl)
    
    return embedding, nil
}
```

### 批量处理

```go
func (m *Matcher) BatchMatch(
    talents []TalentProfile,
    job JobProfile,
) []MatchResult {
    results := make([]MatchResult, len(talents))
    
    // 并发处理
    var wg sync.WaitGroup
    sem := make(chan struct{}, 10) // 限制并发数
    
    for i, talent := range talents {
        wg.Add(1)
        sem <- struct{}{}
        
        go func(idx int, t TalentProfile) {
            defer wg.Done()
            defer func() { <-sem }()
            
            results[idx] = m.Match(t, job)
        }(i, talent)
    }
    
    wg.Wait()
    return results
}
```
