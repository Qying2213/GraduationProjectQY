### Go客户端实现

```go
package coze

import (
    "bytes"
    "context"
    "encoding/json"
    "fmt"
    "io"
    "mime/multipart"
    "net/http"
    "time"
)

type Client struct {
    baseURL    string
    token      string
    workflowID string
    httpClient *http.Client
}

type Config struct {
    BaseURL    string
    Token      string
    WorkflowID string
}

func NewClient(cfg *Config) *Client {
    return &Client{
        baseURL:    cfg.BaseURL,
        token:      cfg.Token,
        workflowID: cfg.WorkflowID,
        httpClient: &http.Client{Timeout: 300 * time.Second},
    }
}

// UploadFile 上传文件获取file_id
func (c *Client) UploadFile(ctx context.Context, filename string, data []byte) (string, error) {
    url := fmt.Sprintf("%s/v1/files/upload", c.baseURL)
    
    var buf bytes.Buffer
    w := multipart.NewWriter(&buf)
    part, _ := w.CreateFormFile("file", filename)
    part.Write(data)
    w.Close()
    
    req, _ := http.NewRequestWithContext(ctx, "POST", url, &buf)
    req.Header.Set("Authorization", "Bearer "+c.token)
    req.Header.Set("Content-Type", w.FormDataContentType())
    
    resp, err := c.httpClient.Do(req)
    if err != nil {
        return "", err
    }
    defer resp.Body.Close()
    
    var result struct {
        Code int `json:"code"`
        Data struct {
            ID string `json:"id"`
        } `json:"data"`
    }
    json.NewDecoder(resp.Body).Decode(&result)
    
    return result.Data.ID, nil
}

// RunWorkflow 执行工作流
func (c *Client) RunWorkflow(ctx context.Context, params map[string]interface{}) (map[string]interface{}, error) {
    url := fmt.Sprintf("%s/v1/workflow/run", c.baseURL)
    
    body := map[string]interface{}{
        "workflow_id":   c.workflowID,
        "response_mode": "blocking",
        "parameters":    params,
    }
    
    jsonBody, _ := json.Marshal(body)
    req, _ := http.NewRequestWithContext(ctx, "POST", url, bytes.NewBuffer(jsonBody))
    req.Header.Set("Authorization", "Bearer "+c.token)
    req.Header.Set("Content-Type", "application/json")
    
    resp, err := c.httpClient.Do(req)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()
    
    var result map[string]interface{}
    json.NewDecoder(resp.Body).Decode(&result)
    
    return result, nil
}
```

### Go客户端实现

```go
package embedding

import (
    "bytes"
    "context"
    "crypto/hmac"
    "crypto/sha256"
    "encoding/hex"
    "encoding/json"
    "fmt"
    "net/http"
    "time"
)

type Client struct {
    endpoint  string
    accessKey string
    secretKey string
    model     string
}

type Config struct {
    Endpoint  string
    AccessKey string
    SecretKey string
    Model     string
}

func NewClient(cfg *Config) *Client {
    return &Client{
        endpoint:  cfg.Endpoint,
        accessKey: cfg.AccessKey,
        secretKey: cfg.SecretKey,
        model:     cfg.Model,
    }
}

type EmbeddingRequest struct {
    Model string   `json:"model"`
    Input []string `json:"input"`
}

type EmbeddingResponse struct {
    Data []struct {
        Embedding []float64 `json:"embedding"`
        Index     int       `json:"index"`
    } `json:"data"`
    Usage struct {
        TotalTokens int `json:"total_tokens"`
    } `json:"usage"`
}

// GetEmbeddings 获取文本向量
func (c *Client) GetEmbeddings(ctx context.Context, texts []string) ([][]float64, error) {
    reqBody := EmbeddingRequest{
        Model: c.model,
        Input: texts,
    }
    
    jsonBody, _ := json.Marshal(reqBody)
    
    req, _ := http.NewRequestWithContext(ctx, "POST", 
        c.endpoint+"/embeddings", bytes.NewBuffer(jsonBody))
    
    // 签名认证
    c.signRequest(req, jsonBody)
    
    req.Header.Set("Content-Type", "application/json")
    
    client := &http.Client{Timeout: 30 * time.Second}
    resp, err := client.Do(req)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()
    
    var result EmbeddingResponse
    if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
        return nil, err
    }
    
    embeddings := make([][]float64, len(result.Data))
    for _, item := range result.Data {
        embeddings[item.Index] = item.Embedding
    }
    
    return embeddings, nil
}

// signRequest 火山引擎签名
func (c *Client) signRequest(req *http.Request, body []byte) {
    timestamp := time.Now().UTC().Format("20060102T150405Z")
    date := timestamp[:8]
    
    // 构建签名字符串
    hashedPayload := sha256Hash(body)
    canonicalRequest := fmt.Sprintf("%s\n%s\n%s\n%s\n%s\n%s",
        req.Method,
        req.URL.Path,
        req.URL.RawQuery,
        "content-type:application/json\nhost:"+req.URL.Host+"\nx-date:"+timestamp,
        "content-type;host;x-date",
        hashedPayload,
    )
    
    stringToSign := fmt.Sprintf("HMAC-SHA256\n%s\n%s/%s/request\n%s",
        timestamp, date, "ark", sha256Hash([]byte(canonicalRequest)))
    
    // 计算签名
    kDate := hmacSHA256([]byte("VOLC"+c.secretKey), date)
    kService := hmacSHA256(kDate, "ark")
    kSigning := hmacSHA256(kService, "request")
    signature := hex.EncodeToString(hmacSHA256(kSigning, stringToSign))
    
    // 设置认证头
    req.Header.Set("X-Date", timestamp)
    req.Header.Set("Authorization", fmt.Sprintf(
        "HMAC-SHA256 Credential=%s/%s/ark/request, SignedHeaders=content-type;host;x-date, Signature=%s",
        c.accessKey, date, signature))
}

func sha256Hash(data []byte) string {
    h := sha256.Sum256(data)
    return hex.EncodeToString(h[:])
}

func hmacSHA256(key []byte, data string) []byte {
    h := hmac.New(sha256.New, key)
    h.Write([]byte(data))
    return h.Sum(nil)
}
```

### 向量相似度计算

```go
package embedding

import "math"

// CosineSimilarity 计算余弦相似度
func CosineSimilarity(a, b []float64) float64 {
    if len(a) != len(b) || len(a) == 0 {
        return 0
    }
    
    var dotProduct, normA, normB float64
    for i := range a {
        dotProduct += a[i] * b[i]
        normA += a[i] * a[i]
        normB += b[i] * b[i]
    }
    
    if normA == 0 || normB == 0 {
        return 0
    }
    
    return dotProduct / (math.Sqrt(normA) * math.Sqrt(normB))
}

// EuclideanDistance 计算欧氏距离
func EuclideanDistance(a, b []float64) float64 {
    if len(a) != len(b) {
        return math.MaxFloat64
    }
    
    var sum float64
    for i := range a {
        diff := a[i] - b[i]
        sum += diff * diff
    }
    
    return math.Sqrt(sum)
}
```
