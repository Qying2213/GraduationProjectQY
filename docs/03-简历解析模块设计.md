## 模块概述

### 文本提取

#### PDF文本提取
使用 `github.com/ledongthuc/pdf` 库提取PDF文本：
```go
func ExtractPDFText(filePath string) (string, error) {
    f, r, err := pdf.Open(filePath)
    if err != nil {
        return "", err
    }
    defer f.Close()
    
    var buf bytes.Buffer
    for i := 1; i <= r.NumPage(); i++ {
        page := r.Page(i)
        text, _ := page.GetPlainText(nil)
        buf.WriteString(text)
    }
    return buf.String(), nil
}
```

#### 图片OCR识别
使用 `github.com/otiai10/gosseract/v2` 进行本地OCR：
```go
func OCRImage(imagePath string) (string, error) {
    client := gosseract.NewClient()
    defer client.Close()
    
    client.SetLanguage("chi_sim", "eng")
    client.SetImage(imagePath)
    
    return client.Text()
}
```

### 风控规则引擎

#### 学历验真
```go
func ValidateEducation(resume *ParsedResume) []RiskItem {
    var risks []RiskItem
    
    // 检查学历时间合理性
    if resume.GraduationYear > 0 {
        age := time.Now().Year() - resume.BirthYear
        eduYears := resume.GraduationYear - resume.BirthYear
        
        if eduYears < 18 || eduYears > 35 {
            risks = append(risks, RiskItem{
                Type:    "education_time",
                Level:   "warning",
                Message: "学历时间存在异常",
            })
        }
    }
    
    return risks
}
```

#### 时间冲突检测
```go
func DetectTimeConflict(workHistory []WorkExperience) []RiskItem {
    var risks []RiskItem
    
    for i := 0; i < len(workHistory)-1; i++ {
        current := workHistory[i]
        next := workHistory[i+1]
        
        // 检查时间重叠
        if current.EndDate.After(next.StartDate) {
            risks = append(risks, RiskItem{
                Type:    "time_conflict",
                Level:   "high",
                Message: fmt.Sprintf("工作经历时间冲突: %s 与 %s", 
                    current.Company, next.Company),
            })
        }
    }
    
    return risks
}
```

## 数据模型

```go
type ParsedResume struct {
    // 基本信息
    Name       string `json:"name"`
    Phone      string `json:"phone"`
    Email      string `json:"email"`
    Location   string `json:"location"`
    
    // 教育背景
    Education      string `json:"education"`
    School         string `json:"school"`
    Major          string `json:"major"`
    GraduationYear int    `json:"graduation_year"`
    
    // 工作经验
    Experience  string           `json:"experience"`
    WorkHistory []WorkExperience `json:"work_history"`
    
    // 技能
    Skills []string `json:"skills"`
    
    // 风控结果
    RiskItems []RiskItem `json:"risk_items"`
    RiskScore int        `json:"risk_score"`
}

type WorkExperience struct {
    Company   string    `json:"company"`
    Position  string    `json:"position"`
    StartDate time.Time `json:"start_date"`
    EndDate   time.Time `json:"end_date"`
    Desc      string    `json:"description"`
}

type RiskItem struct {
    Type    string `json:"type"`
    Level   string `json:"level"`
    Message string `json:"message"`
}
```
